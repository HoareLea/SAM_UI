using System;
using System.Diagnostics;
using System.Linq;
using SAM.Analytical;
using SAM.Analytical.Tas;
using SAM.Core;

internal static class Program
{
    private static void Report(string label, LayerThicknessCalculationData data, LayerThicknessCalculationResult r, long ms)
    {
        Console.WriteLine("{0}: layerIdx(in)={1} U={2} dir={3} ext={4} range={5}-{6} | result: {7} | {8} ms",
            label, data.LayerIndex, data.ThermalTransmittance, data.HeatFlowDirection, data.External, data.ThicknessRange?.Min, data.ThicknessRange?.Max,
            r == null ? "NULL" : string.Format("layerIdx={0} thickness={1} initialU={2} calcU={3} targetU={4}", r.LayerIndex, r.Thickness, r.InitialThermalTransmittance, r.CalculatedThermalTransmittance, r.ThermalTransmittance), ms);
    }

    private static int Main(string[] args)
    {
        AnalyticalModel model = SAM.Core.Convert.ToSAM<AnalyticalModel>(args[0])?.FirstOrDefault();
        ConstructionManager cm = model.ConstructionManager;
        string name = args.Length > 1 ? args[1] : "SIM_EXT_SLD";
        Construction construction = cm.GetConstructions(name).First();
        Console.WriteLine("construction {0}: {1}", name, string.Join(" / ", construction.ConstructionLayers.Select(l => l.Name + " " + l.Thickness)));
        foreach (var l in construction.ConstructionLayers)
        {
            var m = cm.MaterialLibrary.GetMaterial(l.Name);
            Console.WriteLine("   {0} -> {1} {2} k={3}", l.Name, m?.GetType().Name, Core_Type(m), (m as SAM.Core.Material)?.ThermalConductivity);
        }
        ThermalTransmittanceCalculator calc = new ThermalTransmittanceCalculator(cm);
        Func<string, Action<LayerThicknessCalculationData>, bool> run = (label, mutate) =>
        {
            LayerThicknessCalculationData data = SAM.Analytical.Tas.Create.LayerThicknessCalculationData(construction, cm.MaterialLibrary);
            data.ThermalTransmittance = 0.5;
            mutate?.Invoke(data);
            Stopwatch sw = Stopwatch.StartNew();
            var r = calc.Calculate(data) as LayerThicknessCalculationResult;
            Report(label, data, r, sw.ElapsedMilliseconds);
            return true;
        };
        run("default U=0.5", null);
        run("default U=0.2", d => d.ThermalTransmittance = 0.2);
        run("default U=0.05", d => d.ThermalTransmittance = 0.05);
        run("default U=0.05 range 0.001-0.5", d => { d.ThermalTransmittance = 0.05; d.ThicknessRange = new Range<double>(0.001, 0.5); });
        run("default U=0.9 (above initial 0.26.. within range?)", d => d.ThermalTransmittance = 0.9);
        run("default U=5 (too high)", d => d.ThermalTransmittance = 5);
        run("heat flow Undefined", d => d.HeatFlowDirection = HeatFlowDirection.Undefined);
        run("layerIndex -1 forced (fallback)", d => d.LayerIndex = -1);
        for (int i = 0; i < construction.ConstructionLayers.Count; i++) { int k = i; run("explicit layer " + k, d => d.LayerIndex = k); }
        return 0;
    }

    private static string Core_Type(IMaterial m) { return m == null ? "null" : SAM.Core.Query.MaterialType(m).ToString(); }
}
