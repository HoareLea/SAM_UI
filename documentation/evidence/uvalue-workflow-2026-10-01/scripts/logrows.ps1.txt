param([string]$Out = 'C:\TasOut\uvalue\before-ev', [string]$Like = 'Log', [string]$File = 'modelcheck.txt')
. (Join-Path $PSScriptRoot 'ulib.ps1')
$w = Top $Like; $root = $AE::FromHandle($w.Handle)
$g = Find-ById $root 'DataGrid_Main'
$lines = @()
foreach ($r in $g.FindAll($TS::Children, (New-Object System.Windows.Automation.PropertyCondition($AE::ControlTypeProperty, $CT::DataItem)))) {
    $cells = @($r.FindAll($TS::Children, (New-Object System.Windows.Automation.PropertyCondition($AE::ControlTypeProperty, $CT::Custom))) | ForEach-Object { $_.Current.Name })
    $lines += ($cells -join ' | ')
}
$lines | Out-File (Join-Path $Out $File) -Encoding UTF8
$lines | ForEach-Object { Say $_ }
