param([string]$Out = 'C:\TasOut\uvalue\before-ev', [string]$Like = '*', [string]$Press = '', [string]$ClickBox = '', [int]$Timeout = 180, [string]$Shot = '', [string]$ExpectTitle = '*')
# Press a button (or OK on a message box), then time until a window other than the pressed one (and the main window) appears.
. (Join-Path $PSScriptRoot 'ulib.ps1')
if ($ClickBox) {
    $b = Windows | Where-Object { $_.Class -eq '#32770' } | Select-Object -First 1
    if (-not $b) { Say 'FAIL no message box'; exit 1 }
    ScreenShot $Shot
    Say ("box text: {0}" -f (BoxText $b))
    $before = @(Windows | ForEach-Object { $_.Handle.ToInt64() })
    $btn = Get-Children $b.Handle | Where-Object { $_.Class -eq 'Button' } | Select-Object -First 1
    [void][W.N]::SendMessage($btn.Handle, 0x00F5, [System.IntPtr]::Zero, $null)
    $pressed = $b
} else {
    $pressed = Windows | Where-Object { $_.Title -like $Like -and $_.Title -notlike 'SAM Analytical*' } | Select-Object -First 1
    if (-not $pressed) { Say "FAIL window $Like"; exit 1 }
    $before = @(Windows | ForEach-Object { $_.Handle.ToInt64() })
    Say "press $Press in '$($pressed.Title)'"
    Detached ("-File `"$PSScriptRoot\invoke.ps1`" -ProcessId $script:ProcessId -Handle {0} -Id {1}" -f $pressed.Handle.ToInt64(), $Press)
}
$sw = [Diagnostics.Stopwatch]::StartNew()
$k = $pressed.Handle.ToInt64()
while ($sw.Elapsed.TotalSeconds -lt $Timeout) {
    Start-Sleep -Milliseconds 100
    foreach ($w in (Windows)) {
        $h = $w.Handle.ToInt64()
        if ($h -eq $k -or $w.Title -like 'SAM Analytical*') { continue }
        if ($w.Title -like 'Calculating*' -or $w.Title -eq 'Calculate') { continue }
        if ($before -contains $h -and $w.Class -ne '#32770') { continue }
        if ($w.Class -ne '#32770' -and $w.Title -notlike $ExpectTitle) { continue }
        Say ("NEW WINDOW after {0:N1}s: '{1}' [{2}]" -f $sw.Elapsed.TotalSeconds, $w.Title, $w.Class)
        if ($w.Class -eq '#32770') { Say ("  box text: {0}" -f (BoxText $w)) }
        Start-Sleep -Milliseconds 600
        if ($Shot) { if ($w.Class -eq '#32770') { ScreenShot ($Shot + '-box') } else { Shot $w $Shot } }
        exit 0
    }
}
Say "TIMEOUT after $Timeout s"; ListWindows
