param([string]$Out = 'C:\TasOut\uvalue\before-ev', [string]$Like = 'Constructions', [string]$Row = 'SIM_EXT_SLD', [string]$Item = '')
. (Join-Path $PSScriptRoot 'ulib.ps1')
$w = Top $Like; $root = $AE::FromHandle($w.Handle)
$g = Find-ById $root 'DataGrid_Constructions'
foreach ($r in $g.FindAll($TS::Children, (New-Object System.Windows.Automation.PropertyCondition($AE::ControlTypeProperty, $CT::DataItem)))) {
    $c = $r.FindFirst($TS::Children, (New-Object System.Windows.Automation.PropertyCondition($AE::NameProperty, $Row)))
    if (-not $c) { continue }
    $cb = $r.FindFirst($TS::Descendants, (New-Object System.Windows.Automation.PropertyCondition($AE::ControlTypeProperty, $CT::ComboBox)))
    $v = $null; $vp = $null; if ($cb.TryGetCurrentPattern([System.Windows.Automation.ValuePattern]::Pattern, [ref]$vp)) { $v = $vp.Current.Value }
    Say ("row {0}: combo value '{1}'" -f $Row, $v)
    if ($Item -ne '') { Say (Select-ComboItem $cb $Item) } else {
        $xp = $null; if ($cb.TryGetCurrentPattern([System.Windows.Automation.ExpandCollapsePattern]::Pattern, [ref]$xp)) { $xp.Expand(); Start-Sleep -Milliseconds 500 }
        $items = $cb.FindAll($TS::Descendants, (New-Object System.Windows.Automation.PropertyCondition($AE::ControlTypeProperty, $CT::ListItem)))
        Say ("items: " + (($items | ForEach-Object { $_.Current.Name }) -join ' | '))
        if ($xp) { $xp.Collapse() }
    }
}
