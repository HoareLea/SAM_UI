"""Reports, per model, each signal SAM's PartOBaselineFindings refuses on (SAM/SAM.Analytical/Query/PartOBaselineFindings.cs):
run output  - OverheatingScenarios, SimulationResultProvenance, IResult objects, cluster DesignDay records;
materialised - Part O MVHR ventilation systems, per-space Part F internal conditions '<condition> - <space>',
               PartOMaterialisationRecord, PartOIsolationContext."""
import sys, zipfile, re, json, collections
def load(p):
    z = zipfile.ZipFile(p)
    return ''.join(z.read(n).decode('utf-8', 'replace') for n in z.namelist())
for p in sys.argv[1:]:
    t = load(p)
    types = collections.Counter(re.findall(r'"_type":\s*"SAM\.[A-Za-z.]*\.([A-Za-z]+),', t))
    print('==', p)
    print('  run output:')
    print('    Overheating Scenarios parameter:', '"Overheating Scenarios"' in t, '(%d scenario objects)' % types['OverheatingScenario'])
    print('    SimulationResultProvenance:', types['SimulationResultProvenance'])
    res = {k: v for k, v in types.items() if k.endswith('SimulationResult') or k.endswith('Result') and k not in ('SimulationResultProvenance',)}
    print('    result objects:', sum(res.values()), dict(res))
    print('    DesignDay objects:', types['DesignDay'])
    print('  materialised:')
    vs = re.findall(r'"_type":\s*"SAM\.Analytical\.VentilationSystem,SAM\.Analytical"[^{}]*?"Name":\s*"([^"]*)"', t)
    vst = re.findall(r'"_type":\s*"SAM\.Analytical\.VentilationSystemType,SAM\.Analytical"[^{}]*?"Name":\s*"([^"]*)"', t)
    print('    ventilation systems:', vs, ' types:', sorted(set(vst)))
    spaces = set(re.findall(r'"_type":\s*"SAM\.Analytical\.Space,SAM\.Analytical"[^{}]*?"Name":\s*"([^"]*)"', t))
    ics = re.findall(r'"_type":\s*"SAM\.Analytical\.InternalCondition,SAM\.Analytical"[^{}]*?"Name":\s*"([^"]*)"', t)
    per_space = sorted({n for n in ics if any(n.endswith(' - ' + s) for s in spaces)})
    print('    per-space internal conditions (%d):' % len(per_space), per_space[:10])
    print('    PartOMaterialisationRecord:', types['PartOMaterialisationRecord'], ' PartOIsolationContext:', types['PartOIsolationContext'])
