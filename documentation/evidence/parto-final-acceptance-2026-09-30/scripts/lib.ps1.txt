function Say($m) { $l = "{0:HH:mm:ss} {1}" -f (Get-Date), $m; Write-Host $l; Add-Content -Path $log -Value $l -Encoding UTF8 }
function Windows() {
    $list = New-Object System.Collections.Generic.List[object]
    $cb = [W.N+EnumProc]{ param($h, $l)
        [uint32]$p = 0; [void][W.N]::GetWindowThreadProcessId($h, [ref]$p)
        if ($p -eq $script:ProcessId -and [W.N]::IsWindowVisible($h)) {
            $c = New-Object System.Text.StringBuilder 256; [void][W.N]::GetClassName($h, $c, 256)
            $t = New-Object System.Text.StringBuilder 512; [void][A.N]::InternalGetWindowText($h, $t, 512)
            $list.Add([pscustomobject]@{ Handle = $h; Class = $c.ToString(); Title = $t.ToString() })
        }
        return $true }
    [void][W.N]::EnumWindows($cb, [System.IntPtr]::Zero)
    return $list
}
function Detached([string]$Arguments) { [void](Invoke-CimMethod -ClassName Win32_Process -MethodName Create -Arguments @{ CommandLine = "powershell.exe -NoProfile -ExecutionPolicy Bypass -WindowStyle Hidden $Arguments" }) }
function Press($top, [string]$Id) { Say "  press $Id in '$($top.Title)'"; Detached ("-File `"$sp\invoke.ps1`" -ProcessId $script:ProcessId -Handle {0} -Id {1}" -f $top.Handle.ToInt64(), $Id) }
# Waits until a window has gone (closed or hidden), so a closing window is never mistaken for the next one.
function Gone($top, [int]$Seconds = 60) { $k = $top.Handle.ToInt64(); $dl = (Get-Date).AddSeconds($Seconds); while ((Get-Date) -lt $dl) { if (-not (@(Windows) | Where-Object { $_.Handle.ToInt64() -eq $k })) { return $true }; Start-Sleep -Milliseconds 300 }; Say "  (window '$($top.Title)' still open after $Seconds s)"; return $false }
function PressName($top, [string]$Name) { Say "  press '$Name' in '$($top.Title)'"; Detached ("-File `"$sp\invoke.ps1`" -ProcessId $script:ProcessId -Handle {0} -Name `"{1}`"" -f $top.Handle.ToInt64(), $Name) }
# Capture + a text dump of everything UIA can see in the window (names of text, buttons, cells, headers).
function Shot($top, [string]$Name, [switch]$NoDump) {
    $script:seq++
    $base = Join-Path $shots ("{0:00}-{1}" -f $script:seq, $Name)
    $s = [A.N]::Capture($top.Handle, "$base.png")
    Say ("  capture {0:00}-{1} '{2}' ({3})" -f $script:seq, $Name, $top.Title, $s)
    if (-not $NoDump) {
        try { $el = $AE::FromHandle($top.Handle); (Dump $el 0 30) | Out-File "$base.txt" -Encoding UTF8 } catch { "dump failed: $($_.Exception.Message)" | Out-File "$base.txt" }
    }
}
function MsgText($top) { try { $el = $AE::FromHandle($top.Handle); (@($el.FindAll($TS::Descendants, (New-Object System.Windows.Automation.PropertyCondition($AE::ControlTypeProperty, $CT::Text))) | ForEach-Object { $_.Current.Name }) -join ' / ') -replace "`r?`n", ' ' } catch { '?' } }
# Waits for a window whose title matches one of $Like; captures (once) any other new window seen meanwhile,
# e.g. the progress window, and returns the match. A message box (#32770) always counts as a match.
function Await([string[]]$Like, [int]$Seconds = 180, [string]$Tag = 'wait') {
    $seen = @{}; foreach ($w in (Windows)) { $seen[$w.Handle.ToInt64()] = 1 }
    $dl = (Get-Date).AddSeconds($Seconds)
    while ((Get-Date) -lt $dl) {
        Start-Sleep -Milliseconds 400
        foreach ($w in (Windows)) {
            $k = $w.Handle.ToInt64()
            if ($w.Class -eq '#32770') { return $w }
            foreach ($l in $Like) { if ($w.Title -like $l) { return $w } }
            if (-not $seen.ContainsKey($k) -and $w.Title -and $w.Class -notlike '*Popup*' -and $w.Class -ne 'tooltips_class32') {
                $seen[$k] = 1; Start-Sleep -Milliseconds 700; Shot $w "$Tag-transient"
            }
        }
    }
    Say ("  TIMEOUT waiting for {0}; windows: {1}" -f ($Like -join ','), ((Windows | ForEach-Object { "'$($_.Title)'[$($_.Class)]" }) -join ', '))
    return $null
}

function Val($root, $Id) { $e = Find-ById $root $Id; $p = $null; if ($e -and $e.TryGetCurrentPattern([System.Windows.Automation.ValuePattern]::Pattern, [ref]$p)) { $p.Current.Value } else { '<absent>' } }
function HubEl($hub) { $AE::FromHandle($hub.Handle) }
function Txt($root, $Id) { $e = Find-ById $root $Id; if ($e) { $e.Current.Name -replace "`r?`n", ' / ' } else { '<absent>' } }
function Texts($root, $Id) { $e = Find-ById $root $Id; if (-not $e) { return '<absent>' }; (@($e.FindAll($TS::Descendants, (New-Object System.Windows.Automation.PropertyCondition($AE::ControlTypeProperty, $CT::Text))) | Where-Object { -not $_.Current.IsOffscreen } | ForEach-Object { $_.Current.Name }) -join ' | ') -replace "`r?`n", ' / ' }
function Expander($root, $Id, [bool]$Open) { $e = Find-ById $root $Id; $p = $null; if ($e -and $e.TryGetCurrentPattern([System.Windows.Automation.ExpandCollapsePattern]::Pattern, [ref]$p)) { if ($Open) { $p.Expand() } else { $p.Collapse() }; Start-Sleep -Milliseconds 900; "$Id $($p.Current.ExpandCollapseState)" } else { "${Id}: no expander" } }
function Toggle($root, $Id) { $e = Find-ById $root $Id; $p = $null; if ($e -and $e.TryGetCurrentPattern([System.Windows.Automation.TogglePattern]::Pattern, [ref]$p)) { $p.Toggle(); Start-Sleep -Milliseconds 900; "$Id $($p.Current.ToggleState)" } else { "${Id}: no toggle" } }
function HubSummary($hub, [string]$When) {
    $h = HubEl $hub
    Say "--- HUB: $When"
    Say ("  scenario line: " + (Txt $h 'textBlock_Scenario'))
    Say ("  scope: " + (Txt $h 'textBlock_Scope'))
    Say ("  strip: " + (Texts $h 'stepStrip'))
    Say ("  last outcome: " + (Txt $h 'textBlock_LastOutcome'))
    Say ("  status: " + (Texts $h 'itemsControl_Status'))
    Say ("  blockers: " + (Txt $h 'textBlock_Blockers'))
    Say ("  next: " + (Txt $h 'textBlock_NextStep'))
    foreach ($id in 'button_Run', 'button_Review', 'button_Optimise') { $e = Find-ById $h $id; if ($e) { Say ("  {0}: '{1}' enabled={2} tooltip='{3}'" -f $id, $e.Current.Name, $e.Current.IsEnabled, ($e.Current.HelpText -replace "`r?`n", ' / ')) } }
    Say ("  It3: " + (Texts $h 'border_Iteration3'))
}

# A WinForms/Win32 message box: its text from the Static child, answered by BM_CLICK on the button's dialog id.
function BoxText($top) { (@(Get-Children $top.Handle | Where-Object { $_.Class -eq 'Static' } | ForEach-Object { $t = New-Object System.Text.StringBuilder 4096; [void][W.N]::GetWindowText($_.Handle, $t, 4096); $t.ToString() } | Where-Object { $_ }) -join ' / ') -replace "`r?`n", ' ' }
function BoxAnswer($top, [string]$Answer) {
    $ids = @{ 'OK' = 1; 'Cancel' = 2; 'Yes' = 6; 'No' = 7 }
    $b = Get-Children $top.Handle | Where-Object { $_.Class -eq 'Button' -and $_.Id -eq $ids[$Answer] } | Select-Object -First 1
    if (-not $b -and $Answer -eq 'No') { Say '  REFUSING to answer: no No button'; return }
    if (-not $b) { $b = Get-Children $top.Handle | Where-Object { $_.Class -eq 'Button' } | Select-Object -First 1 }
    Say "  answer '$Answer' (button id $($b.Id))"; [void][W.N]::SendMessage($b.Handle, 0x00F5, [System.IntPtr]::Zero, $null)
}
# Handles whatever a Hub action led to until the Hub is back: captures every window, answers a message box
# with $MsgAnswer, and runs $OnWindow for a named window (scriptblock receiving the window).
function Follow([hashtable]$Handlers, [string]$MsgAnswer = 'OK', [int]$Seconds = 240, [string]$Tag = 'step') {
    $dl = (Get-Date).AddSeconds($Seconds)
    while ((Get-Date) -lt $dl) {
        $w = Await (@('Part O*Prepare &*') + @($Handlers.Keys)) ([int]($dl - (Get-Date)).TotalSeconds) $Tag
        if ($null -eq $w) { return $null }
        if ($w.Title -like 'Part O*Prepare &*' -and $w.Class -ne '#32770') { Start-Sleep -Seconds 2; return $w }
        if ($w.Class -eq '#32770') {
            Start-Sleep -Milliseconds 500; Say ("  MESSAGE BOX '{0}': {1}" -f $w.Title, (BoxText $w)); Shot $w "$Tag-messagebox"
            BoxAnswer $w $MsgAnswer; [void](Gone $w 20); continue
        }
        foreach ($k in $Handlers.Keys) { if ($w.Title -like $k) { Start-Sleep -Seconds 2; & $Handlers[$k] $w; break } }
        Start-Sleep -Seconds 2
    }
    return $null
}
