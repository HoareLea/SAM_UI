param([string]$PressDetached = '', [string]$Window = 'Part O*Mixed Design*', [int]$Seconds = 1800, [string]$Tag = 'run', [string]$StopWhenFinalLike = '')
# Presses a control (detached), then logs every window of the process until the progress windows are gone and
# the Mixed Design window is back (or a message box appears, which is captured and left for the caller).
. (Join-Path $PSScriptRoot 'mlib.ps1')
if ($PressDetached) { $w = Top $Window; Press $w $PressDetached }
$t0 = Get-Date; $seen = @{}; $quiet = 0; $lastTas = ''
$dl = $t0.AddSeconds($Seconds)
while ((Get-Date) -lt $dl) {
    Start-Sleep -Seconds 3
    $ws = @(Windows)
    foreach ($x in $ws) {
        $k = $x.Handle.ToInt64()
        if (-not $seen.ContainsKey($k)) {
            $seen[$k] = 1
            if ($x.Title -notlike 'SAM Analytical*') { Say ("  [+{0:0}s] window '{1}' ({2})" -f ((Get-Date) - $t0).TotalSeconds, $x.Title, $x.Class); Start-Sleep -Milliseconds 800; Shot $x ("$Tag-" + (($x.Title -replace '[^A-Za-z0-9]+', '-').Trim('-'))) }
        }
        if ($x.Class -eq '#32770') { Say ("  MESSAGE BOX '{0}': {1}" -f $x.Title, (BoxText $x)); return }
    }
    $tp = TasProcs; if ($tp -ne $lastTas) { Say ("  [+{0:0}s] TAS processes: '{1}'" -f ((Get-Date) - $t0).TotalSeconds, $tp); $lastTas = $tp }
    $others = @($ws | Where-Object { $_.Title -notlike 'SAM Analytical*' -and $_.Title -notlike 'Part O*Mixed Design*' })
    if ($others.Count -eq 0 -and ((Get-Date) - $t0).TotalSeconds -gt 10) { $quiet++ } else { $quiet = 0 }
    if ($quiet -ge 3) { break }
}
Say ("  watch '{0}' ended after {1:0} s" -f $Tag, ((Get-Date) - $t0).TotalSeconds)
