"""Cross-case check of the stage trees written by runs.ps1 (path|size|ticks|sha256 relative to the PartO root)."""
import io, sys, os
E = 'C:/TasOut/parto-final-acceptance-2026-09-30/'
stages = sys.argv[1:] or ['after-1a', 'after-1a-it3', 'after-1b', 'after-2', 'after-2-it3', 'after-close']
def load(n):
    d = {}
    for l in io.open(E + 'tree-%s.txt' % n, encoding='utf-8-sig'):
        l = l.strip()
        if l:
            p, size, ticks, h = l.split('|')
            d[p] = (size, ticks, h)
    return d
trees = [(s, load(s)) for s in stages]
out = []
prev = {}
for name, t in trees:
    new = sorted(set(t) - set(prev))
    changed = sorted(p for p in set(t) & set(prev) if t[p] != prev[p])
    removed = sorted(set(prev) - set(t))
    out.append('== %s: %d files, %d new, %d changed, %d removed' % (name, len(t), len(new), len(changed), len(removed)))
    cases = sorted({p.split('\\')[0] for p in new})
    out.append('   new files in: %s' % ', '.join(cases))
    for p in changed:
        tag = 'content same, write time moved' if t[p][2] == prev[p][2] else 'CONTENT CHANGED'
        out.append('   changed: %s (%s)' % (p, tag))
    for p in removed:
        out.append('   removed: %s' % p)
    prev = t
final = trees[-1][1]
loose = [p for p in final if '\\' not in p]
out.append('== loose files directly in the root: %d %s' % (len(loose), loose))
bad = []
for p in final:
    parts = p.split('\\')
    if len(parts) == 2 and parts[1] != 'PartOCase.json':
        bad.append(p)
    if len(parts) >= 3:
        sub, fn = parts[1], parts[-1].lower()
        if fn.endswith('.timing.csv') or fn.endswith('-operatingairflow.csv'):
            if sub != 'diagnostics': bad.append(p)
        elif fn.endswith('-tm59.txt') or '-iteration3-' in fn:
            if sub != 'reports': bad.append(p)
        elif os.path.splitext(fn)[1] in ('.xml', '.t3d', '.tbd', '.tpd', '.tsd', '.sam', '.json'):
            if sub != 'tas': bad.append(p)
        else:
            bad.append(p)
out.append('== misplaced files: %d %s' % (len(bad), bad))
out.append('== case folders: %s' % sorted({p.split('\\')[0] for p in final}))
print('\n'.join(out))
