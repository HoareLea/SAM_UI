$script:failures = New-Object System.Collections.Generic.List[string]
function Check([bool]$Ok, [string]$What) { if ($Ok) { Say "  CHECK OK   $What" } else { Say "  CHECK FAIL $What"; $script:failures.Add($What) } }
function Abort([string]$m) { Say "ABORT $m"; Say ("windows: " + ((Windows | ForEach-Object { "'$($_.Title)'[$($_.Class)]" }) -join ', ')); Tree 'at-abort'; Stop-Process -Id $script:ProcessId -Force -ErrorAction SilentlyContinue; exit 1 }
function BoxAnswer($top, [string]$Answer) {
    $buttons = @(Get-Children $top.Handle | Where-Object { $_.Class -eq 'Button' })
    if ($buttons.Count -ne 1) { Abort ("a {0}-button dialog appeared: {1}" -f $buttons.Count, (BoxText $top)) }
    $script:failures.Add("message box: " + (BoxText $top))
    Say "  answer the single button"; [void][W.N]::SendMessage($buttons[0].Handle, 0x00F5, [System.IntPtr]::Zero, $null)
}
function Tree([string]$When) {
    $f = Join-Path $Out "tree-$When.txt"
    @(Get-ChildItem $Root -Recurse -File -ErrorAction SilentlyContinue | Sort-Object FullName | ForEach-Object { "{0}|{1}|{2}|{3}" -f $_.FullName.Substring($Root.Length + 1), $_.Length, $_.LastWriteTimeUtc.Ticks, (Get-FileHash $_.FullName -Algorithm SHA256).Hash }) | Set-Content -Encoding UTF8 $f
    Say ("  tree '{0}': {1} file(s) -> {2}" -f $When, @(Get-Content $f).Count, (Split-Path $f -Leaf))
}
function CopyAll($w, [string]$File) {
    Set-Clipboard -Value ' '
    Press $w 'button_CopyAll'
    $dl = (Get-Date).AddSeconds(20); $text = $null
    do { Start-Sleep -Milliseconds 700; $text = Get-Clipboard -Raw } while (((-not $text) -or $text.Trim().Length -eq 0) -and (Get-Date) -lt $dl)
    Set-Content -Path (Join-Path $Out $File) -Value $text -Encoding UTF8
    Say ("  copied {0} chars to {1}" -f $text.Length, $File)
    return $text
}

function RunScenario([string]$Tag, [string]$Like) {
    $h = HubEl $script:hub
    Say ("scenario {0}: {1}" -f $Tag, (Select-ComboItem (Find-ById $h 'comboBox_Scenario') $Like)); Start-Sleep -Seconds 3
    [void](Expander $h 'expander_SimulationCase' $true)
    $box = Find-ById $h 'textBox_OutputDirectory'
    if ($Tag -eq '1a') { Say ("  type the Part O root: " + (Set-Text $box $Root)); Start-Sleep -Seconds 1 }
    $shown = $box.Current.Name; $vp = $null; if ($box.TryGetCurrentPattern([System.Windows.Automation.ValuePattern]::Pattern, [ref]$vp)) { $shown = $vp.Current.Value }
    Say ("  output folder box: '{0}' tooltip='{1}'" -f $shown, $box.Current.HelpText)
    Check ($shown -eq $Root) "$Tag output folder box holds the chosen root exactly"
    HubSummary $script:hub "$Tag before run"
    Shot $script:hub "$Tag-hub-before-run"
    Press $script:hub 'button_Run'; [void](Gone $script:hub)
    $script:tm59Tag = $Tag
    $handlers = @{
        'Part O*Review iteration*' = { param($w) Say ("  REVIEW [{0}] {1}" -f $Tag, (Txt ($AE::FromHandle($w.Handle)) 'textBlock_Summary')); Shot $w "$Tag-review"; Press $w 'button_Accept'; [void](Gone $w) }
        'Part O*CIBSE TM59*' = $tm59Handler
        'Part O*Overheating*' = $tm59Handler
        'Log' = { param($w) $t = MsgText $w; Say ("  LOG WINDOW [{0}]: {1}" -f $Tag, $t); $script:failures.Add("$Tag Log window: $t"); Shot $w "$Tag-log"; [void][W.N]::PostMessage($w.Handle, 0x0010, [System.IntPtr]::Zero, [System.IntPtr]::Zero); [void](Gone $w) }
    }
    $t0 = Get-Date
    $script:hub = Follow $handlers 'OK' 2400 $Tag
    if (-not $script:hub) { Abort "hub did not come back after $Tag" }
    Say ("{0} run + TM59 done in {1:N1} min" -f $Tag, ((Get-Date) - $t0).TotalMinutes)
    HubSummary $script:hub "$Tag after run"
    Shot $script:hub "$Tag-hub-after-run"
    Tree "after-$Tag"
}
function MethodRadios($h) {
    $list = @()
    foreach ($id in 'stackPanel_It3Methods', 'stackPanel_It3MethodsAdvanced') {
        $m = Find-ById $h $id; if (-not $m) { continue }
        foreach ($rb in @($m.FindAll($TS::Descendants, (New-Object System.Windows.Automation.PropertyCondition($AE::ControlTypeProperty, $CT::RadioButton))))) {
            $si = $null; [void]$rb.TryGetCurrentPattern([System.Windows.Automation.SelectionItemPattern]::Pattern, [ref]$si)
            $lbl = (@($rb.FindAll($TS::Descendants, (New-Object System.Windows.Automation.PropertyCondition($AE::ControlTypeProperty, $CT::Text))) | ForEach-Object { $_.Current.Name }) -join ' / ')
            $list += [pscustomobject]@{ El = $rb; Label = $lbl; Selected = $si.Current.IsSelected; Panel = $id }
        }
    }
    return $list
}
function Comparison($w, [string]$Tag) {
    $r = $AE::FromHandle($w.Handle)
    foreach ($id in 'textBlock_Heading', 'textBlock_Cases', 'textBlock_TileReference', 'textBlock_TileSystem', 'textBlock_Refusal') {
        $e = Find-ById $r $id; $v = if ($e -and -not $e.Current.IsOffscreen) { $e.Current.Name -replace "`r?`n", ' / ' } else { '<hidden/absent>' }; Say ("  [{0}] {1}: {2}" -f $Tag, $id, $v)
    }
    $ref = Find-ById $r 'border_Refusal'
    Check (-not ($ref -and -not $ref.Current.IsOffscreen -and ((Txt $r 'textBlock_Refusal') -match '\S'))) "$Tag comparison shows no refusal"
    Shot $w "$Tag-comparison"
    [void](CopyAll $w "$Tag-comparison.txt")
    Press $w 'button_Close'; [void](Gone $w)
}
function RunIteration3([string]$Tag) {
    $h = HubEl $script:hub
    Say ("  It3 reference: " + (Txt $h 'textBlock_It3Reference'))
    $radios = MethodRadios $h
    foreach ($x in $radios) { Say ("  method [{0}] '{1}' selected={2}" -f $x.Panel, $x.Label, $x.Selected) }
    $target = if ($script:method) { $radios | Where-Object { $_.Label -eq $script:method } | Select-Object -First 1 } else { $radios | Where-Object { $_.Selected } | Select-Object -First 1 }
    if (-not $target) { $target = $radios | Select-Object -First 1 }
    if ($target.Panel -eq 'stackPanel_It3MethodsAdvanced') { [void](Expander $h 'expander_It3Advanced' $true) }
    if (-not $target.Selected) { Say ("  select method: " + (Invoke-El $target.El)); Start-Sleep -Seconds 4 }
    $script:method = $target.Label
    $b3 = Find-ById $h 'button_Iteration3'
    Say ("  button_Iteration3 '{0}' enabled={1}; hint: {2}; refusal: {3}" -f $b3.Current.Name, $b3.Current.IsEnabled, (Txt $h 'textBlock_It3Hint'), (Txt $h 'textBlock_It3Refusal'))
    Shot $script:hub "$Tag-it3-hub-before-run"
    if (-not $b3.Current.IsEnabled) { Abort ("Run Iteration 3 not available for method '{0}'" -f $script:method) }
    Press $script:hub 'button_Iteration3'; [void](Gone $script:hub)
    $t0 = Get-Date
    $script:hub = Follow @{ 'Part O*Iteration 3 comparison*' = { param($w) Say ("{0} Iteration 3 finished in {1:N1} min" -f $Tag, ((Get-Date) - $t0).TotalMinutes); Comparison $w "$Tag-it3" } } 'OK' 5400 "$Tag-it3"
    if (-not $script:hub) { Abort "hub did not come back after $Tag Iteration 3" }
    HubSummary $script:hub "$Tag after Iteration 3"
    Tree "after-$Tag-it3"
}
