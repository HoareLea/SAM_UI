param([string]$Model)
# Final acceptance (30 Sep 2026): open Mixed Design on the given model, record whether SAM accepts it as a clean
# baseline (the full baseline text, readiness, Build state), capture it, close the window WITHOUT any action, quit.
# Closing the window writes <model>.partomixed.json beside the model by design (RunPartOMixedDesignCommand).
$sc = $PSScriptRoot
& "$sc\mlaunch.ps1" -Model $Model
& "$sc\mopenmixed.ps1"
. "$sc\mlib.ps1"
$w = Top 'Part O*Mixed Design*'
if (-not $w) { Say 'FAIL Mixed Design window not open'; exit 1 }
$r = $AE::FromHandle($w.Handle)
Say ("MIXED PROBE model {0}" -f $Model)
Say ("  BASELINE: " + (Txt $r 'textBlock_Baseline'))
Say ("  NEXT: " + (Txt $r 'textBlock_Next'))
Say ("  READINESS: " + (Txt $r 'textBlock_Readiness'))
foreach ($id in 'button_Check', 'button_Build', 'button_Save', 'button_SetNatural', 'button_SetMvhr') { Say ("  {0}: {1}" -f $id, (En $r $id)) }
$e = Find-ById $r 'textBlock_Baseline'; if ($e) { Say ("  baseline tooltip: " + ($e.Current.HelpText -replace "`r?`n", ' / ')) }
Shot $w 'mixed-probe'
Press $w 'button_Close'; Start-Sleep -Seconds 3
foreach ($x in (Windows)) { Say ("  window after close '{0}' [{1}]" -f $x.Title, $x.Class) }
Stop-Process -Id $script:ProcessId -Force -ErrorAction SilentlyContinue; Start-Sleep -Seconds 3
Say 'MIXED PROBE DONE'
