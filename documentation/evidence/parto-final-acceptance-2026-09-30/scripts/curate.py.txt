"""Copies the curated evidence of the final acceptance into the SAM_UI repo (no TAS result files)."""
import os, shutil, io
E = 'C:/TasOut/parto-final-acceptance-2026-09-30/'
D = '<SAM-BIM>/SAM_UI/documentation/evidence/parto-final-acceptance-2026-09-30/'
S = '<SAM_daily>/2026-09-29 partOi/'
os.makedirs(D + 'shots', exist_ok=True); os.makedirs(D + 'scripts', exist_ok=True)
def cp(src, dst):
    shutil.copyfile(src, D + dst)
# journeys and watches (.log is gitignored in SAM_UI -> .txt)
cp(E + 'journey-runs.log', 'journey-runs.txt')
for c in ('Iteration1a', 'Iteration1b', 'Iteration2'):
    cp(E + 'journey-reopen-%s.log' % c, 'journey-reopen-%s.txt' % c)
    cp(E + 'tas-processes-reopen-%s.txt' % c, 'tas-processes-reopen-%s.txt' % c)
cp(E + 'tas-processes-runs.txt', 'tas-processes-runs.txt')
cp(E + 'mixed/drive.log', 'journey-mixed.txt')
# trees, checks, snapshots (work area only; historical snapshots summarised in snapshot-diffs.txt)
for n in ('after-1a', 'after-1a-it3', 'after-1b', 'after-2', 'after-2-it3'):
    cp(E + 'tree-%s.txt' % n, 'tree-%s.txt' % n)
for n in ('crosscheck-runs.txt', 'pathscan-runs.txt', 'baselinescan.txt', 'snapshot-diffs.txt'):
    cp(E + n, n)
cp(E + 'snap-work-before-runs.txt', 'snapshot-work-before.txt')
cp(E + 'snap-work-after-mixed-probe.txt', 'snapshot-work-final.txt')
cp(E + 'snap-work-before.txt', 'snapshot-work-initial.txt')
# window text captured with Copy All
for n in ('1a-tm59-window.txt', '1b-tm59-window.txt', '2-tm59-window.txt', '1a-it3-it3-comparison.txt', '2-it3-it3-comparison.txt',
          'reopen-Iteration1a-comparison.txt', 'reopen-Iteration2-comparison.txt'):
    cp(E + n, 'window-' + n)
# the PartOCase markers, verbatim
with io.open(D + 'PartOCase-markers.txt', 'w', encoding='utf-8', newline='\n') as f:
    for c in ('Iteration1a', 'Iteration1b', 'Iteration2', 'Iteration3'):
        f.write('%s/PartOCase.json: %s\n' % (c, io.open(S + 'PartO/%s/PartOCase.json' % c, encoding='utf-8-sig').read().strip()))
# screenshots that prove UI behaviour
shots = {
    'shots-runs/01-hub-at-open.png': '01-hub-at-open-stale-provenance.png',
    'shots-runs/02-1a-hub-before-run.png': '02-hub-root-typed.png',
    'shots-runs/08-1a-it3-it3-comparison.png': '03-it3-vs-1a-comparison.png',
    'shots-runs/21-2-it3-it3-comparison.png': '04-it3-vs-2-comparison.png',
    'shots-reopen-Iteration1a/05-Iteration1a-reopen-comparison.png': '05-reopen-it3-vs-1a-no-tas.png',
    'shots-reopen-Iteration2/05-Iteration2-reopen-comparison.png': '06-reopen-it3-vs-2-no-tas.png',
    'mixed/shots/02-baseline-refused.png': '07-mixed-refused-source-model.png',
    'mixed/shots/04-mixed-probe.png': '08-mixed-refused-iteration2-result.png',
}
for s, d in shots.items():
    cp(E + s, 'shots/' + d)
for n in os.listdir(E + 'scripts'):
    if n.endswith(('.ps1', '.py')) and not n.startswith('adapt_'):
        cp(E + 'scripts/' + n, 'scripts/' + n + '.txt')
print('\n'.join(sorted(os.path.relpath(os.path.join(dp, f), D) for dp, dn, fs in os.walk(D) for f in fs)))
