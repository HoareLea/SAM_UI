// U-value PR2a engine probe: the PUBLIC engine API on a COPY of the model with real Tas.
// UValueViewModel + TasUValueEvaluator (STA single-flight, debounce) -> CreateRequest -> Modify.SetUValue
// (real Tas.Modify.UpdateThermalParameters) -> ModelCheck (AnalyticalModel.Log) -> Undo -> Redo.
using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Linq;
using System.Threading;
using System.Threading.Tasks;
using System.Windows.Threading;
using SAM.Analytical;
using SAM.Analytical.UI;
using SAM.Analytical.UI.WPF;
using SAM.Core;

internal static class Program
{
    private static readonly Stopwatch clock = Stopwatch.StartNew();

    private static void Log(string format, params object[] args)
    {
        Console.WriteLine("[{0,6} ms] {1}", clock.ElapsedMilliseconds, args.Length == 0 ? format : string.Format(format, args));
    }

    private static void Pump(Func<bool> until, int timeoutMs)
    {
        Stopwatch sw = Stopwatch.StartNew();
        while (!until() && sw.ElapsedMilliseconds < timeoutMs)
        {
            DispatcherFrame frame = new DispatcherFrame();
            Dispatcher.CurrentDispatcher.BeginInvoke(DispatcherPriority.Background, new Action(() => frame.Continue = false));
            Dispatcher.PushFrame(frame);
            Thread.Sleep(10);
        }
    }

    private static void Settle(UValueViewModel vm, int timeoutMs = 20000)
    {
        Thread.Sleep(50);
        Pump(() => !vm.IsBusy && vm.Status != UValuePreviewStatus.Calculating, timeoutMs);
    }

    private static void Show(string label, UValueViewModel vm)
    {
        Log("{0}: target='{1}' status={2} busy={3} current U={4:0.####} achieved={5:0.####} thickness={6:0.0} mm margin={7:0.####} eval={8} ms apply={9}",
            label, vm.TargetText, vm.Status, vm.IsBusy, vm.CurrentThermalTransmittance, vm.CalculatedThermalTransmittance, vm.CalculatedThickness * 1000, vm.Margin, vm.LastEvaluationMilliseconds, vm.ApplyEnabled);
        if (vm.StatusMessage != null) Log("   message: {0}", vm.StatusMessage);
        if (vm.ApplyBlockReason != null) Log("   blocked: {0}", vm.ApplyBlockReason);
    }

    private static List<LogRecord> Errors(AnalyticalModel model)
    {
        return model.Log().ToList().FindAll(x => x.LogRecordType == LogRecordType.Error);
    }

    [STAThread]
    private static int Main(string[] args)
    {
        Console.OutputEncoding = System.Text.Encoding.UTF8;
        string path = args[0];
        string pathAfter = args[1];
        SynchronizationContext.SetSynchronizationContext(new DispatcherSynchronizationContext(Dispatcher.CurrentDispatcher));

        AnalyticalModel model = SAM.Core.Convert.ToSAM<AnalyticalModel>(path).First();
        Construction source = model.AdjacencyCluster.GetConstructions().First(x => x.Name == "SIM_EXT_SLD");
        List<Panel> using_ = model.AdjacencyCluster.GetPanels(source);
        List<Guid> selected = using_.Take(3).Select(x => x.Guid).ToList();
        Log("model {0}: SIM_EXT_SLD {1} used by {2} panels ({3}); selected 3", path, source.Guid, using_.Count, string.Join(", ", using_.GroupBy(x => x.PanelType).Select(g => g.Count() + " " + g.Key)));

        List<LogRecord> errorsBefore = Errors(model);
        Log("ModelCheck before: {0} error(s)", errorsBefore.Count);
        foreach (LogRecord r in errorsBefore) Log("   E {0}", r.Text);

        using TasUValueEvaluator evaluator = new TasUValueEvaluator();
        using UValueViewModel vm = new UValueViewModel(model, source.Guid, selected, evaluator);

        Task init = vm.InitializeAsync();
        Pump(() => init.IsCompleted, 20000);
        Settle(vm);
        Show("open", vm);
        Log("   layer: {0}", vm.LayerSentence);
        Log("   scope: {0}", vm.ScopeText);
        Log("   result: {0}", vm.ResultText);
        Log("   basis: {0} {1} external={2} fromPanels={3} mixed={4}", vm.HeatFlowBasis.PanelType, vm.HeatFlowDirection, vm.External, vm.HeatFlowBasis.FromPanels, vm.HeatFlowBasis.Mixed);
        Log("   range: U {0:0.####} at 1 mm .. U {1:0.####} at 1000 mm", vm.MinThicknessThermalTransmittance, vm.MaxThicknessThermalTransmittance);
        foreach (string w in vm.Warnings) Log("   warning: {0}", w);

        // Typing "0.3" key by key, 80 ms apart: the debounce must run one evaluation for the last text only.
        foreach (string text in new[] { "0", "0.", "0.3" })
        {
            vm.TargetText = text;
            Pump(() => false, 80);
        }
        Settle(vm);
        Show("typed 0.3", vm);
        foreach (UValueLayerRow row in vm.PreviewRows) Log("   row {0} {1}: {2} -> {3} mm{4}", row.Index, row.Name, row.ThicknessBefore, row.ThicknessAfter, row.Adjusted ? "  <- adjusted" : "");

        foreach (string text in new[] { "0.01", "5", "0.05", "0.5", "0.3" })
        {
            Stopwatch sw = Stopwatch.StartNew();
            vm.TargetText = text;
            Settle(vm);
            Show(string.Format("typed {0} (settled after {1} ms incl. 300 ms debounce)", text, sw.ElapsedMilliseconds), vm);
        }

        SetUValueRequest request = vm.CreateRequest();
        Log("request: layer {0} thickness {1:0.0000} m U {2:0.###} -> {3:0.###} mode {4} scope {5} name '{6}'", request.LayerIndex, request.Thickness, request.InitialThermalTransmittance, request.CalculatedThermalTransmittance, request.Mode, request.Scope, request.NewConstructionName);

        UIAnalyticalModel ui = new UIAnalyticalModel(model);
        int modified = 0, history = 0;
        ui.Modified += (s, e) => modified++;
        ui.HistoryChanged += (s, e) => history++;
        Stopwatch applyWatch = Stopwatch.StartNew();
        SetUValueResult result = ui.SetUValue(request);
        Log("apply ({0} ms incl. Tas UpdateThermalParameters): succeeded={1} error={2} Modified events={3} HistoryChanged={4} CanUndo={5}", applyWatch.ElapsedMilliseconds, result.Succeeded, result.Error, modified, history, ui.CanUndo);
        Log("   new construction '{0}' ({1}); material {2} -> {3} (added={4}); {5:0.000} -> {6:0.000} m; U {7:0.###} -> {8:0.###}; {9} panels",
            result.Construction.Name, result.Construction.Guid, result.SourceMaterialName, result.MaterialName, result.MaterialAdded, result.OldThickness, result.NewThickness, result.OldThermalTransmittance, result.NewThermalTransmittance, result.PanelCount);

        AnalyticalModel after = ui.JSAMObject;
        Log("   material library has {0}: {1}", result.MaterialName, after.MaterialLibrary.GetMaterial(result.MaterialName) != null);
        Log("   panels on new: {0}; panels on source: {1}; source still in model: {2}", after.AdjacencyCluster.GetPanels(result.Construction).Count, after.AdjacencyCluster.GetPanels(source).Count, after.AdjacencyCluster.GetConstructions().Any(x => x.Guid == source.Guid));
        List<LogRecord> errorsAfter = Errors(after);
        Log("ModelCheck after apply: {0} error(s)", errorsAfter.Count);
        foreach (LogRecord r in errorsAfter) Log("   E {0}", r.Text);
        SAM.Core.Convert.ToFile(after, pathAfter);
        Log("saved {0}", pathAfter);

        // Undo / Redo (async restore marshalled back through the dispatcher).
        bool undoStarted = ui.Undo();
        Pump(() => ui.CanRedo, 30000);
        AnalyticalModel undone = ui.JSAMObject;
        Log("undo started={0}: CanUndo={1} CanRedo={2}; panels on source {3}/{4}; new construction present={5}; material present={6}; materials {7} (original {8})",
            undoStarted, ui.CanUndo, ui.CanRedo, undone.AdjacencyCluster.GetPanels(source).Count, using_.Count,
            undone.AdjacencyCluster.GetConstructions().Any(x => x.Guid == result.Construction.Guid), undone.MaterialLibrary.GetMaterial(result.MaterialName) != null,
            undone.MaterialLibrary.GetMaterials().Count, model.MaterialLibrary.GetMaterials().Count);

        bool redoStarted = ui.Redo();
        Pump(() => ui.CanUndo && !ui.CanRedo, 30000);
        AnalyticalModel redone = ui.JSAMObject;
        Log("redo started={0}: panels on new {1}; material present={2}", redoStarted, redone.AdjacencyCluster.GetPanels(result.Construction).Count, redone.MaterialLibrary.GetMaterial(result.MaterialName) != null);
        return 0;
    }
}
