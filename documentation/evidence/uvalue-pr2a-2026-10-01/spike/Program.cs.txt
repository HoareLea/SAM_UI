// U-value PR2a timing spike: real ThermalTransmittanceCalculator on a COPY of the model.
// A: 50 consecutive reachable evaluations on one STA thread.  B: unreachable timings.
// C: best achievable U at the max thickness via Calculate(Guid) (one TCD call).
// D: STA single-flight worker with debounce + stale-result cancellation (typing bursts).
// E: two concurrent evaluations on two STA threads (is single-flight needed?).
using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.IO;
using System.Linq;
using System.Threading;
using SAM.Analytical;
using SAM.Analytical.Tas;
using SAM.Core;

internal sealed class SingleFlightStaWorker : IDisposable
{
    private readonly object gate = new object();
    private readonly AutoResetEvent signal = new AutoResetEvent(false);
    private readonly Thread thread;
    private readonly int debounceMs;
    private readonly Func<double, LayerThicknessCalculationResult> evaluate;
    private long latestVersion;
    private double latestTarget;
    private DateTime latestSubmitUtc;
    private bool disposed;

    public int Started, Completed, Discarded;
    public event Action<long, double, LayerThicknessCalculationResult, long> Delivered;

    public SingleFlightStaWorker(int debounceMs, Func<double, LayerThicknessCalculationResult> evaluate)
    {
        this.debounceMs = debounceMs;
        this.evaluate = evaluate;
        thread = new Thread(Loop) { IsBackground = true, Name = "uvalue-sta" };
        thread.SetApartmentState(ApartmentState.STA);
        thread.Start();
    }

    public long Submit(double target)
    {
        lock (gate)
        {
            latestVersion++;
            latestTarget = target;
            latestSubmitUtc = DateTime.UtcNow;
        }
        signal.Set();
        return latestVersion;
    }

    private void Loop()
    {
        long done = 0;
        while (true)
        {
            signal.WaitOne();
            while (true)
            {
                long version; double target; DateTime submitted;
                lock (gate)
                {
                    if (disposed) return;
                    version = latestVersion; target = latestTarget; submitted = latestSubmitUtc;
                }
                if (version == done) break;
                int wait = debounceMs - (int)(DateTime.UtcNow - submitted).TotalMilliseconds;
                if (wait > 0) { Thread.Sleep(wait); lock (gate) { if (latestVersion != version) continue; } }
                Interlocked.Increment(ref Started);
                Stopwatch sw = Stopwatch.StartNew();
                LayerThicknessCalculationResult result = evaluate(target);
                long ms = sw.ElapsedMilliseconds;
                Interlocked.Increment(ref Completed);
                done = version;
                bool stale;
                lock (gate) { stale = latestVersion != version; }
                if (stale) { Interlocked.Increment(ref Discarded); continue; }
                Delivered?.Invoke(version, target, result, ms);
            }
        }
    }

    public void Dispose()
    {
        lock (gate) { disposed = true; }
        signal.Set();
        thread.Join(5000);
    }
}

internal static class Program
{
    private static string Fmt(LayerThicknessCalculationResult r)
    {
        return r == null ? "NULL" : string.Format("layer={0} t={1:0.0}mm U0={2} U={3} target={4}", r.LayerIndex, r.Thickness * 1000, r.InitialThermalTransmittance, r.CalculatedThermalTransmittance, r.ThermalTransmittance);
    }

    private static T OnSta<T>(Func<T> func)
    {
        T value = default;
        Thread t = new Thread(() => value = func());
        t.SetApartmentState(ApartmentState.STA);
        t.Start();
        t.Join();
        return value;
    }

    private static string Stats(List<long> ms)
    {
        List<long> s = ms.OrderBy(x => x).ToList();
        return string.Format("n={0} min={1} median={2} p90={3} max={4} mean={5:0} ms", s.Count, s[0], s[s.Count / 2], s[(int)(s.Count * 0.9)], s[s.Count - 1], s.Average());
    }

    [STAThread]
    private static int Main(string[] args)
    {
        string path = args[0];
        AnalyticalModel model = SAM.Core.Convert.ToSAM<AnalyticalModel>(path)?.FirstOrDefault();
        ConstructionManager cm = model.ConstructionManager;
        Construction construction = cm.GetConstructions("SIM_EXT_SLD").First();
        ThermalTransmittanceCalculator calc = new ThermalTransmittanceCalculator(cm);
        string tempDir = Path.Combine(Path.GetTempPath(), "SAM");
        int tcdBefore = Directory.Exists(tempDir) ? Directory.GetFiles(tempDir, "*.tcd").Length : 0;
        Console.WriteLine("Main thread apartment: {0}", Thread.CurrentThread.GetApartmentState());

        Func<double, LayerThicknessCalculationResult> eval = target =>
        {
            LayerThicknessCalculationData data = SAM.Analytical.Tas.Create.LayerThicknessCalculationData(construction, cm.MaterialLibrary);
            data.ThermalTransmittance = target;
            return calc.Calculate(data);
        };

        // Warm-up (first COM activation).
        Stopwatch w = Stopwatch.StartNew();
        LayerThicknessCalculationResult warm = OnSta(() => eval(0.5));
        Console.WriteLine("warm-up U=0.5: {0} | {1} ms", Fmt(warm), w.ElapsedMilliseconds);

        // A: 50 consecutive reachable evaluations on one STA thread.
        Console.WriteLine("\n== A: 50 consecutive reachable evaluations (one STA thread)");
        List<long> a = OnSta(() =>
        {
            List<long> list = new List<long>();
            for (int i = 0; i < 50; i++)
            {
                double target = Math.Round(0.15 + (i % 10) * 0.025 + (i / 10) * 0.003, 3); // 0.15..0.39, all below U0=0.26? mixed
                Stopwatch sw = Stopwatch.StartNew();
                LayerThicknessCalculationResult r = eval(target);
                list.Add(sw.ElapsedMilliseconds);
                Console.WriteLine("  #{0:00} target {1:0.000}: {2} | {3} ms", i + 1, target, Fmt(r), sw.ElapsedMilliseconds);
            }
            return list;
        });
        Console.WriteLine("A: " + Stats(a));

        // B: unreachable.
        Console.WriteLine("\n== B: unreachable targets");
        List<long> b = OnSta(() =>
        {
            List<long> list = new List<long>();
            foreach (double target in new[] { 5.0, 5.0, 5.0, 0.02, 0.02, 0.02 })
            {
                Stopwatch sw = Stopwatch.StartNew();
                LayerThicknessCalculationResult r = eval(target);
                list.Add(sw.ElapsedMilliseconds);
                Console.WriteLine("  target {0}: {1} | {2} ms", target, Fmt(r), sw.ElapsedMilliseconds);
            }
            return list;
        });
        Console.WriteLine("B: " + Stats(b));

        // C: best achievable U at the range ends via Calculate(Guid) on a copy of the construction.
        Console.WriteLine("\n== C: U at the range ends (one TCD call each, Calculate(Guid))");
        int layerIndex = SAM.Analytical.Tas.Query.AdjustableLayerIndex(construction, cm.MaterialLibrary);
        foreach (double thickness in new[] { 1.0, 0.001, 0.08 })
        {
            List<ConstructionLayer> layers = construction.ConstructionLayers;
            layers[layerIndex] = new ConstructionLayer(layers[layerIndex].Name, thickness);
            Construction probe = new Construction(Guid.NewGuid(), construction, construction.Name + " [probe]");
            probe = new Construction(probe, layers);
            ConstructionManager cm2 = new ConstructionManager(cm);
            cm2.Add(probe);
            ThermalTransmittanceCalculator calc2 = new ThermalTransmittanceCalculator(cm2);
            Stopwatch sw = Stopwatch.StartNew();
            ThermalTransmittanceCalculationResult r = OnSta(() => calc2.Calculate(new[] { probe.Guid })?.FirstOrDefault());
            Console.WriteLine("  layer {0} at {1} mm: U(horizontal, external) = {2} | {3} ms", layerIndex, thickness * 1000, r?.GetThermalTransmittance(HeatFlowDirection.Horizontal, true), sw.ElapsedMilliseconds);
        }

        // D: single-flight worker, typing bursts.
        Console.WriteLine("\n== D: STA single-flight worker, debounce 300 ms");
        using (SingleFlightStaWorker worker = new SingleFlightStaWorker(300, eval))
        {
            List<string> delivered = new List<string>();
            Stopwatch clock = Stopwatch.StartNew();
            long lastSubmitMs = 0;
            ManualResetEventSlim got = new ManualResetEventSlim(false);
            worker.Delivered += (v, t, r, ms) =>
            {
                delivered.Add(string.Format("v{0} target {1}: {2} (eval {3} ms, {4} ms after last keystroke)", v, t, Fmt(r), ms, clock.ElapsedMilliseconds - lastSubmitMs));
                got.Set();
            };

            // D1: typing "0.3" -> "0.35" -> "0.355" at 150 ms per key (only the last should run).
            foreach (double t in new[] { 0.3, 0.35, 0.355 }) { worker.Submit(t); lastSubmitMs = clock.ElapsedMilliseconds; Thread.Sleep(150); }
            got.Wait(10000); got.Reset(); Thread.Sleep(500);
            Console.WriteLine("  D1 typing burst: started={0} completed={1} discarded={2}", worker.Started, worker.Completed, worker.Discarded);

            // D2: new keystrokes while an evaluation is in flight (the in-flight result must be discarded).
            worker.Submit(0.05); lastSubmitMs = clock.ElapsedMilliseconds;          // slow-ish (469 mm)
            Thread.Sleep(450);                                                     // debounce over, evaluation running
            foreach (double t in new[] { 0.4, 0.45 }) { worker.Submit(t); lastSubmitMs = clock.ElapsedMilliseconds; Thread.Sleep(120); }
            got.Wait(15000); got.Reset(); Thread.Sleep(800);
            Console.WriteLine("  D2 burst during in-flight: started={0} completed={1} discarded={2}", worker.Started, worker.Completed, worker.Discarded);

            // D3: unreachable then quickly a reachable target.
            worker.Submit(5); lastSubmitMs = clock.ElapsedMilliseconds; Thread.Sleep(400);
            worker.Submit(0.3); lastSubmitMs = clock.ElapsedMilliseconds;
            got.Wait(15000); Thread.Sleep(800);
            Console.WriteLine("  D3 unreachable then reachable: started={0} completed={1} discarded={2}", worker.Started, worker.Completed, worker.Discarded);
            foreach (string line in delivered) Console.WriteLine("    delivered " + line);
        }

        // E: two concurrent evaluations on two STA threads.
        Console.WriteLine("\n== E: two concurrent evaluations (two STA threads, separate calculators)");
        LayerThicknessCalculationResult e1 = null, e2 = null; long m1 = 0, m2 = 0;
        Thread t1 = new Thread(() => { Stopwatch sw = Stopwatch.StartNew(); e1 = eval(0.3); m1 = sw.ElapsedMilliseconds; });
        Thread t2 = new Thread(() => { Stopwatch sw = Stopwatch.StartNew(); LayerThicknessCalculationData d = SAM.Analytical.Tas.Create.LayerThicknessCalculationData(construction, cm.MaterialLibrary); d.ThermalTransmittance = 0.4; e2 = new ThermalTransmittanceCalculator(cm).Calculate(d); m2 = sw.ElapsedMilliseconds; });
        t1.SetApartmentState(ApartmentState.STA); t2.SetApartmentState(ApartmentState.STA);
        t1.Start(); t2.Start();
        bool j1 = t1.Join(60000), j2 = t2.Join(60000);
        Console.WriteLine("  t1 joined={0} {1} | {2} ms", j1, Fmt(e1), m1);
        Console.WriteLine("  t2 joined={0} {1} | {2} ms", j2, Fmt(e2), m2);

        int tcdAfter = Directory.Exists(tempDir) ? Directory.GetFiles(tempDir, "*.tcd").Length : 0;
        Console.WriteLine("\nleftover temp .tcd files: before={0} after={1}", tcdBefore, tcdAfter);
        return 0;
    }
}
