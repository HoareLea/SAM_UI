// PR4 measurement: which piece of SAM_Tas's results read ("Adding Results") grows with the project.
// Usage: tsdprobe <path.tsd> [zones-to-time]
using SAM.Analytical.Tas;
using SAM.Core.Tas;
using System.Diagnostics;
using TSD;

internal static class Program
{
    [STAThread]
    private static void Main(string[] args)
    {
        string path_TSD = args[0];
        Stopwatch total = Stopwatch.StartNew();
        void Log(string text) => Console.WriteLine("{0,8:0.0} s  {1}", total.Elapsed.TotalSeconds, text);

        using SAMTSDDocument sAMTSDDocument = new(path_TSD, true);
        SimulationData simulationData = sAMTSDDocument.TSDDocument.SimulationData;
        Log("opened " + path_TSD);

        Stopwatch step = Stopwatch.StartNew();
        BuildingData buildingData = simulationData.GetBuildingData();
        List<ZoneData> zoneDatas = SAM.Analytical.Tas.Query.ZoneDatas(buildingData);
        Log(string.Format("ZoneDatas: {0} zones in {1:0.00} s", zoneDatas.Count, step.Elapsed.TotalSeconds));

        step.Restart();
        var cooling = SAM.Analytical.Tas.Query.ValueDictionary(buildingData, tsdZoneArray.coolingLoad);
        SAM.Analytical.Tas.Query.ValueDictionary(buildingData, tsdZoneArray.heatingLoad);
        Log(string.Format("ValueDictionary x2: {0:0.00} s", step.Elapsed.TotalSeconds));

        step.Restart();
        var designData = SAM.Analytical.Tas.Query.DesignDataDictionary(simulationData);
        Log(string.Format("DesignDataDictionary: {0} in {1:0.00} s", designData?.Count, step.Elapsed.TotalSeconds));

        int count = args.Length > 1 && int.TryParse(args[1], out int value) && value > 0 ? Math.Min(value, zoneDatas.Count) : zoneDatas.Count;
        TimeSpan overheating = TimeSpan.Zero, surfaces = TimeSpan.Zero, designZone = TimeSpan.Zero, peakGroup = TimeSpan.Zero;
        for (int i = 0; i < count; i++)
        {
            ZoneData zoneData = zoneDatas[i];

            step.Restart();
            SAM.Analytical.Tas.Query.Overheating(zoneData, simulationData.firstDay, simulationData.lastDay);
            overheating += step.Elapsed;

            step.Restart();
            if (designData != null && designData.TryGetValue(zoneData.zoneGUID, out var tuple) && tuple?.Item1 is CoolingDesignData coolingDesignData)
            {
                coolingDesignData.GetZoneData(zoneData.zoneNumber);
            }
            designZone += step.Elapsed;

            step.Restart();
            int index = cooling != null && cooling.TryGetValue(zoneData.zoneGUID, out Tuple<double, int> peak) ? peak.Item2 : 1;
            zoneData.ToSAM_SurfaceSimulationResults(Math.Max(index, 1));
            surfaces += step.Elapsed;

            step.Restart();
            buildingData.TryGetMax(new[] { zoneData.zoneGUID }, tsdZoneArray.coolingLoad, out _, out _);
            peakGroup += step.Elapsed;
        }

        Log(string.Format("per zone over {0} zones (ms/zone): Overheating {1:0.0}, design-day zone {2:0.0}, surface results {3:0.0}, GetPeakZoneGroupGains {4:0.0}", count, overheating.TotalMilliseconds / count, designZone.TotalMilliseconds / count, surfaces.TotalMilliseconds / count, peakGroup.TotalMilliseconds / count));

        if (args.Length > 2) return;
        step.Restart();
        var results = SAM.Analytical.Tas.Convert.ToSAM_Results(simulationData);
        Log(string.Format("Convert.ToSAM_Results (whole): {0} results in {1:0.0} s", results?.Count, step.Elapsed.TotalSeconds));
    }
}
