$script:failures = New-Object System.Collections.Generic.List[string]
function Check([bool]$Ok, [string]$What) { if ($Ok) { Say "  CHECK OK   $What" } else { Say "  CHECK FAIL $What"; $script:failures.Add($What) } }
function Abort([string]$m) { Say "ABORT $m"; Stop-Process -Id $script:ProcessId -Force -ErrorAction SilentlyContinue; exit 1 }
function BoxAnswer($top, [string]$Answer) {
    $buttons = @(Get-Children $top.Handle | Where-Object { $_.Class -eq 'Button' })
    if ($buttons.Count -ne 1) { Abort ("a {0}-button dialog appeared: {1}" -f $buttons.Count, (BoxText $top)) }
    $script:failures.Add("message box: " + (BoxText $top))
    Say "  answer the single button"; [void][W.N]::SendMessage($buttons[0].Handle, 0x00F5, [System.IntPtr]::Zero, $null)
}
function CopyAll($w, [string]$File) {
    Set-Clipboard -Value ' '
    Press $w 'button_CopyAll'
    $dl = (Get-Date).AddSeconds(20); $text = $null
    do { Start-Sleep -Milliseconds 700; $text = Get-Clipboard -Raw } while (((-not $text) -or $text.Trim().Length -eq 0) -and (Get-Date) -lt $dl)
    Set-Content -Path (Join-Path $Out $File) -Value $text -Encoding UTF8
    Say ("  copied {0} chars to {1}" -f $text.Length, $File)
    return $text
}
$script:tm59Seen = $false
$tm59Handler = { param($w)
    $script:tm59Seen = $true
    $r = $AE::FromHandle($w.Handle)
    Say ("  TM59 [{0}] '{1}' verdict: {2}" -f $Case, $w.Title, (Txt $r 'textBlock_Verdict'))
    Shot $w "$Case-reopen-tm59"
    $t = CopyAll $w "reopen-$Case-tm59-window.txt"
    Check ($t -match [regex]::Escape((Join-Path $Root "$Case\tas"))) "$Case TM59 window names its results in $Case\tas"
    Press $w 'button_Close'; [void](Gone $w)
}
$script:compSeen = $false
$compHandler = { param($w)
    $script:compSeen = $true
    $r = $AE::FromHandle($w.Handle)
    foreach ($id in 'textBlock_Heading', 'textBlock_Cases', 'textBlock_TileReference', 'textBlock_TileSystem', 'textBlock_Refusal') {
        $e = Find-ById $r $id; $v = if ($e -and -not $e.Current.IsOffscreen) { $e.Current.Name -replace "`r?`n", ' / ' } else { '<hidden/absent>' }; Say ("  [{0}] {1}: {2}" -f $Case, $id, $v)
    }
    $cases = Txt $r 'textBlock_Cases'
    Check ($cases -match 'no TAS simulation was run') "$Case comparison says it was reopened with no TAS simulation"
    if ($Case -eq 'Iteration1a') { Check ($cases -match '-It1a-It3B') "$Case system case carries the -It1a qualifier" } else { Check (-not ($cases -match '-It1a-It3B')) "$Case system case is unqualified" }
    $ref = Find-ById $r 'border_Refusal'
    Check (-not ($ref -and -not $ref.Current.IsOffscreen -and ((Txt $r 'textBlock_Refusal') -match '\S'))) "$Case reopened comparison shows no refusal"
    Shot $w "$Case-reopen-comparison"
    [void](CopyAll $w "reopen-$Case-comparison.txt")
    Press $w 'button_Close'; [void](Gone $w)
}
